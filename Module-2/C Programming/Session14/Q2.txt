// This function checks whether a number is even or odd.
function isEven(num) {

    // Check if the number is divisible by 2.
    // The % operator gives the remainder.
    if (num % 2 == 0) {

        // If the remainder is 0, the number is even.
        return true;

    } else {

        // Otherwise, the number is odd.
        return false;
    }
}
